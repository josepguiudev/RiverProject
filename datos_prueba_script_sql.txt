-- ============================================
-- VERIFICACIÓN FINAL
-- ============================================
SELECT 
    s.id_survey,
    s.name AS encuesta,
    q.id_question,
    q.text_question AS pregunta,
    qc.type_name AS tipo,
    COUNT(o.id_option) AS num_opciones
FROM survey s
JOIN question q ON q.id_survey = s.id_survey
JOIN question_config qc ON qc.id_question = q.id_question
LEFT JOIN options o ON o.id_question = q.id_question
GROUP BY s.id_survey, s.name, q.id_question, q.text_question, qc.type_name
ORDER BY s.id_survey, q.id_question;
