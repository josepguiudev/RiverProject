DECLARE @OptRanger   BIGINT = (SELECT TOP 1 id_option FROM [dbo].[options] WHERE text_opcion = 'Ranger');
DECLARE @OptMago     BIGINT = (SELECT TOP 1 id_option FROM [dbo].[options] WHERE text_opcion = 'Mago');

DECLARE @OptSelva    BIGINT = (SELECT TOP 1 id_option FROM [dbo].[options] WHERE text_opcion = 'Selva');
DECLARE @OptInfra    BIGINT = (SELECT TOP 1 id_option FROM [dbo].[options] WHERE text_opcion = 'Inframundo');
DECLARE @OptHallow   BIGINT = (SELECT TOP 1 id_option FROM [dbo].[options] WHERE text_opcion = 'Hallow');

    -- Vincular a encuesta
    INSERT INTO [dbo].[user_survey] ([id_survey], [id_user], [is_respondida]) VALUES (@SurveyId, @UserId, 1);

    -- Respuesta Q1: NUMERIC (Horas incrementales)

    -- Respuesta Q3: MULTIPLE_CHOICE (Variación de Biomas)
    -- Lógica: Los pares eligen Selva e Inframundo, los múltiplos de 3 Hallow, otros solo Selva
    INSERT INTO [dbo].[respuesta] ([id_user], [id_option], [is_completada]) VALUES (@UserId, @OptSelva, 1);
    
    IF (@i % 2 = 0) 
        INSERT INTO [dbo].[respuesta] ([id_user], [id_option], [is_completada]) VALUES (@UserId, @OptInfra, 1);
    
    IF (@i % 3 = 0)
        INSERT INTO [dbo].[respuesta] ([id_user], [id_option], [is_completada]) VALUES (@UserId, @OptHallow, 1);

    SET @i = @i + 1;
END
